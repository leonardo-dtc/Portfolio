# v5 · the glass edition — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `v5/`, the Liquid Glass / visionOS edition of Leonardo's portfolio, exactly as the approved spec describes. It opens on an out-of-focus room with the name written in liquid glass; Enter leads into one room of glass windows.

**Architecture:** Static HTML pages (one per address, complete without scripts), with one stylesheet and ES-module scripts. A single WebGL2 canvas behind everything draws the room, every glass panel, and the written name. DOM elements carry the content and tell the renderer where their glass is. A small navigation layer swaps window contents in place, so the room never reloads.

**Tech Stack:** HTML, CSS, vanilla JavaScript (ES modules, no build step), WebGL2 (GLSL ES 3.00). Tests: Node 24 `node:test` for pure modules, and playwright-core 1.64 with Chrome for browser checks.

**Spec:** `docs/superpowers/specs/2026-09-24-v5-liquid-glass-design.md`. Comps are in `docs/superpowers/specs/v5-comps/`.

`v5/assets/js/springs.js`:
```js
// Springs as Apple describes them: a response (the period, in seconds) and a damping ratio.
export function createSpring({ value = 0, target = value, response = .5, damping = .86 } = {}) {
  const s = { value, target, velocity: 0, response, damping };
  s.step = (dt) => {
    const w = 2 * Math.PI / s.response, k = w * w, c = 2 * s.damping * w;
    const t = Math.min(dt, .064), n = Math.max(1, Math.ceil(t / .004)), h = t / n;
    for (let i = 0; i < n; i++) { s.velocity += (-k * (s.value - s.target) - c * s.velocity) * h; s.value += s.velocity * h; }
    return Math.abs(s.value - s.target) < 1e-3 && Math.abs(s.velocity) < 1e-3;
  };
  s.snap = (v = s.target) => { s.value = s.target = v; s.velocity = 0; };
  return s;
}
import { onFrame } from './frame.js';
export function tween(spring, target, onUpdate) {
  spring.target = target;
  return new Promise(resolve => {
    const off = onFrame(dt => { const done = spring.step(dt); onUpdate(spring.value); if (done) { spring.snap(target); onUpdate(target); off(); resolve(); } });
  });
}
```

`v5/assets/js/frame.js`:
```js
// One requestAnimationFrame loop for everything that moves, so the glass and its windows share a frame.
const subs = new Set();
let last = 0, running = false;
function loop(now) {
  const dt = last ? Math.min(.05, (now - last) / 1000) : 1 / 60; last = now;
  for (const fn of [...subs]) fn(dt, now / 1000);
  if (subs.size) requestAnimationFrame(loop); else { running = false; last = 0; }
}
export function onFrame(fn) {
  subs.add(fn);
  if (!running && typeof requestAnimationFrame === 'function') { running = true; requestAnimationFrame(loop); }
  return () => subs.delete(fn);
}
```

```js
// Projective maps between a panel's own pixels (0..w, 0..h) and its four corners on screen.
function squareToQuad(q) {
  const [[x0, y0], [x1, y1], [x2, y2], [x3, y3]] = q;
  const sx = x0 - x1 + x2 - x3, sy = y0 - y1 + y2 - y3;
  let g = 0, h = 0;
  if (Math.abs(sx) > 1e-9 || Math.abs(sy) > 1e-9) {
    const dx1 = x1 - x2, dx2 = x3 - x2, dy1 = y1 - y2, dy2 = y3 - y2, den = dx1 * dy2 - dx2 * dy1;
    g = (sx * dy2 - dx2 * sy) / den; h = (dx1 * sy - sx * dy1) / den;
  }
  return [x1 - x0 + g * x1, x3 - x0 + h * x3, x0, y1 - y0 + g * y1, y3 - y0 + h * y3, y0, g, h, 1];
}
export function mul3(a, b) {
  const o = new Array(9);
  for (let r = 0; r < 3; r++) for (let c = 0; c < 3; c++) o[r * 3 + c] = a[r * 3] * b[c] + a[r * 3 + 1] * b[3 + c] + a[r * 3 + 2] * b[6 + c];
  return o;
}
export function rectToQuad(w, h, quad) { return mul3(squareToQuad(quad), [1 / w, 0, 0, 0, 1 / h, 0, 0, 0, 1]); }
export function invert3(m) {
  const [a, b, c, d, e, f, g, h, i] = m;
  const A = e * i - f * h, B = -(d * i - f * g), C = d * h - e * g, det = a * A + b * B + c * C;
  return [A / det, -(b * i - c * h) / det, (b * f - c * e) / det, B / det, (a * i - c * g) / det, -(a * f - c * d) / det, C / det, -(a * h - b * g) / det, (a * e - b * d) / det];
}
export function applyH(m, x, y) { const X = m[0] * x + m[1] * y + m[2], Y = m[3] * x + m[4] * y + m[5], W = m[6] * x + m[7] * y + m[8]; return [X / W, Y / W]; }
```

```glsl
#version 300 es
precision highp float;
uniform sampler2D uScene; uniform sampler2D uInk;
uniform vec2 uRes; uniform float uLod; uniform float uFrostLod; uniform float uTime; uniform float uDay;
uniform vec2 uLight; uniform int uCount;
uniform mat3 uInv[16]; uniform vec4 uBox[16]; uniform vec4 uState[16];
uniform vec4 uInk0; uniform vec3 uInkX;          // on, _, _, _ ; scale, offset x, offset y (css px → mask px)
out vec4 o;
float sdRound(vec2 p, vec2 b, float r) { vec2 q = abs(p) - b + r; return length(max(q, 0.0)) + min(max(q.x, q.y), 0.0) - r; }
vec3 roomAt(vec2 uv, float lod) {
  vec2 t = exp2(lod) / uRes;
  vec3 c = textureLod(uScene, uv, lod).rgb * .36;
  c += textureLod(uScene, uv + vec2(t.x, 0.), lod).rgb * .16; c += textureLod(uScene, uv - vec2(t.x, 0.), lod).rgb * .16;
  c += textureLod(uScene, uv + vec2(0., t.y), lod).rgb * .16; c += textureLod(uScene, uv - vec2(0., t.y), lod).rgb * .16;
  return c;
}
void main() {
  vec2 px = vec2(gl_FragCoord.x, uRes.y - gl_FragCoord.y);          // y down, device px
  vec2 uv = gl_FragCoord.xy / uRes;
  vec3 col = roomAt(uv, uLod);
  float shadow = 0.;
  int hit = -1; vec2 lp = vec2(0.); float sd = 1e5;
  for (int i = 0; i < 16; i++) {
    if (i >= uCount) break;
    vec3 h = uInv[i] * vec3(px, 1.);
    vec2 p = h.xy / h.z, half = uBox[i].xy * .5;
    float d = sdRound(p - half, half, uBox[i].z);
    vec3 hs = uInv[i] * vec3(px - vec2(0., 18.), 1.);
    float ds = sdRound(hs.xy / hs.z - half, half, uBox[i].z);
    shadow = max(shadow, (1. - smoothstep(-20., 70., ds)) * .30 * uState[i].x);
    if (d < 1.5) { hit = i; lp = p; sd = d; }
  }
  col *= 1. - shadow;
  if (hit >= 0) {
    vec2 half = uBox[hit].xy * .5; float r = uBox[hit].z, kind = uBox[hit].w, m = uState[hit].x, dim = uState[hit].y;
    float e = 1.;
    vec2 n = normalize(vec2(sdRound(lp + vec2(e, 0.) - half, half, r) - sdRound(lp - vec2(e, 0.) - half, half, r),
                            sdRound(lp + vec2(0., e) - half, half, r) - sdRound(lp - vec2(0., e) - half, half, r)) + 1e-6);
    float lensW = kind < .5 ? 28. : 18.;
    float edge = clamp(1. + sd / lensW, 0., 1.);
    float bend = edge * edge * (kind < .5 ? 22. : 14.) * m;
    vec2 suv = (px - n * bend) / uRes; suv.y = 1. - suv.y;
    vec3 g = roomAt(suv, max(uLod, mix(uLod, uFrostLod, m)));
    vec3 tint = mix(vec3(.05, .07, .22), vec3(.12, .18, .42), uDay);
    g = mix(g, tint, .34 * m) * (1. - .12 * m);
    vec2 L = normalize(uLight - px + 1e-3);
    float rim = exp(-pow((sd + 1.1) / 1.1, 2.));
    g += vec3(1.) * rim * (.35 + .5 * max(dot(n, -L), 0.)) * .75 * m;
    g += vec3(1.) * edge * edge * .07 * m;
    g += vec3(1.) * exp(-dot(px - uLight, px - uLight) / (2. * 260. * 260.)) * .07 * m;
    g *= 1. - .5 * dim;
    col = mix(col, g, smoothstep(1.5, -.5, sd) * min(1., m * 1.6));
  }
  // the written name (Task 7 fills the ink block)
  o = vec4(col, 1.);
}
```

```js
import { SCENE, COMPOSITE, VERT } from './shaders.js';
export function createRoom(canvas) {
  const gl = canvas.getContext('webgl2', { antialias: false, alpha: false, powerPreference: 'high-performance' });
  if (!gl) return null;
  const prog = (fs) => { const p = gl.createProgram(); for (const [t, s] of [[gl.VERTEX_SHADER, VERT], [gl.FRAGMENT_SHADER, fs]]) { const sh = gl.createShader(t); gl.shaderSource(sh, s); gl.compileShader(sh); if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(sh)); gl.attachShader(p, sh); } gl.linkProgram(p); if (!gl.getProgramParameter(p, gl.LINK_STATUS)) throw new Error(gl.getProgramInfoLog(p)); return p; };
  const sceneP = prog(SCENE), compP = prog(COMPOSITE);
  const vao = gl.createVertexArray(); gl.bindVertexArray(vao);
  gl.bindBuffer(gl.ARRAY_BUFFER, gl.createBuffer()); gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
  gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0, 2, gl.FLOAT, false, 0, 0);
  const U = (p, names) => Object.fromEntries(names.map(n => [n, gl.getUniformLocation(p, n)]));
  const us = U(sceneP, ['uRes', 'uTime', 'uDay', 'uAspect', 'uShift']);
  const uc = U(compP, ['uScene', 'uInk', 'uRes', 'uLod', 'uFrostLod', 'uTime', 'uDay', 'uLight', 'uCount', 'uInv', 'uBox', 'uState', 'uInk0', 'uInkX']);

  let w = 0, h = 0, dpr = Math.min(devicePixelRatio || 1, 1.5), maxLod = 6;
  const st = { defocus: 0, day: 0, shift: [0, 0], light: [0, 0], panels: { count: 0 }, ink: null, fast: 0 };
  function size() {
    const nw = Math.round(innerWidth * dpr), nh = Math.round(innerHeight * dpr);
    if (nw === w && nh === h) return;
    w = canvas.width = nw; h = canvas.height = nh;
    gl.bindTexture(gl.TEXTURE_2D, tex); gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA8, w, h, 0, gl.RGBA, gl.UNSIGNED_BYTE, null);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.MIRRORED_REPEAT); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.MIRRORED_REPEAT);
    gl.bindFramebuffer(gl.FRAMEBUFFER, fbo); gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, tex, 0); gl.bindFramebuffer(gl.FRAMEBUFFER, null);
    maxLod = Math.log2(Math.max(8, .045 * w));        // the hello's defocus: sigma about 4.5% of the width
  }
  let frame = 0, acc = 0, slow = 0, t0 = performance.now();
  const off = onFrame((dt) => {
    if (document.hidden) return;
    frame++;
    const idle = st.fast <= 0; st.fast -= dt;
    if (idle && frame % 2) return;                       // 30 fps when nothing moves
    size();
    const t = (performance.now() - t0) / 1000;
    gl.bindFramebuffer(gl.FRAMEBUFFER, fbo); gl.viewport(0, 0, w, h); gl.useProgram(sceneP);
    gl.uniform2f(us.uRes, w, h); gl.uniform1f(us.uTime, t); gl.uniform1f(us.uDay, st.day); gl.uniform1f(us.uAspect, w / h); gl.uniform2f(us.uShift, st.shift[0], st.shift[1]);
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
    gl.bindTexture(gl.TEXTURE_2D, tex); gl.generateMipmap(gl.TEXTURE_2D);
    gl.bindFramebuffer(gl.FRAMEBUFFER, null); gl.viewport(0, 0, w, h); gl.useProgram(compP);
    gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, tex); gl.uniform1i(uc.uScene, 0);
    gl.activeTexture(gl.TEXTURE1); gl.bindTexture(gl.TEXTURE_2D, inkTex); gl.uniform1i(uc.uInk, 1);
    if (st.ink && st.ink.dirty) { gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, st.ink.canvas); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE); st.ink.dirty = false; }
    gl.uniform2f(uc.uRes, w, h); gl.uniform1f(uc.uLod, st.defocus * maxLod); gl.uniform1f(uc.uFrostLod, Math.log2(Math.max(4, 22 * dpr)));
    gl.uniform1f(uc.uTime, t); gl.uniform1f(uc.uDay, st.day); gl.uniform2f(uc.uLight, st.light[0] * dpr, st.light[1] * dpr);
    const P = st.panels; gl.uniform1i(uc.uCount, P.count || 0);
    if (P.count) { gl.uniformMatrix3fv(uc.uInv, true, P.inv); gl.uniform4fv(uc.uBox, P.box); gl.uniform4fv(uc.uState, P.state); }
    const ink = st.ink; gl.uniform4f(uc.uInk0, ink ? ink.on : 0, 0, 0, 0); gl.uniform3f(uc.uInkX, ...(ink ? ink.xform : [1, 0, 0]));
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
    // budget: after the first 90 frames, a slow machine drops to 1x and stops the room's own motion
    if (frame < 90) { acc += dt; if (frame === 89 && acc / 89 > .022 && dpr > 1) { dpr = 1; slow = 1; } }
    if (slow) t0 += dt * 1000;
  });
  return {
    get dpr() { return dpr; },
    set(o) { Object.assign(st, o); st.fast = Math.max(st.fast, .5); },
    kick(s = 2) { st.fast = Math.max(st.fast, s); },
    destroy() { off(); },
  };
}
```

```js
import { rectToQuad, invert3 } from './geometry.js';
const KIND = { window: 0, ornament: 1, control: 2 };
const inv = new Float32Array(144), box = new Float32Array(64), state = new Float32Array(64);
export function readPanels(root = document, dpr = 1) {
  const els = [...root.querySelectorAll('[data-glass]')].filter(el => el.offsetParent !== null || getComputedStyle(el).position === 'fixed');
  let n = 0;
  for (const el of els) {
    if (n === 16) break;
    const g = el.glass || { m: 1, dim: 0 };
    if (g.m <= .001) continue;
    const cs = getComputedStyle(el);
    if (cs.visibility === 'hidden' || cs.display === 'none') continue;
    let quad;
    const probes = el.querySelectorAll(':scope > .probe');
    const w = el.offsetWidth, h = el.offsetHeight;
    if (probes.length === 4) quad = [...probes].map(p => { const r = p.getBoundingClientRect(); return [r.left * dpr, r.top * dpr]; });
    else { const r = el.getBoundingClientRect(); quad = [[r.left, r.top], [r.right, r.top], [r.right, r.bottom], [r.left, r.bottom]].map(([x, y]) => [x * dpr, y * dpr]); }
    const sx = probes.length === 4 ? w * dpr : (quad[1][0] - quad[0][0]), sy = probes.length === 4 ? h * dpr : (quad[3][1] - quad[0][1]);
    const m = invert3(rectToQuad(sx, sy, quad));
    inv.set(m, n * 9);
    const rad = Math.min(parseFloat(cs.borderTopLeftRadius) || 0, Math.min(w, h) / 2) * dpr * (sx / (w * dpr || 1));
    box.set([sx, sy, rad, KIND[el.dataset.glass] ?? 0], n * 4);
    state.set([g.m, g.dim, 0, 0], n * 4);
    n++;
  }
  return { count: n, inv, box, state };
}
```
